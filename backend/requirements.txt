# Postgres driver. psycopg 3 rather than psycopg2: the venv runs Python 3.14,
# for which psycopg2-binary has no wheel and would need a source build.
# [binary] bundles libpq, so no compiler or libpq-dev is needed on the host.
# 3.3.4 is the first release shipping cp314 wheels — 3.2.x stops at cp313.
psycopg[binary]==3.3.4

# bcrypt.__about__, removed upstream), so the app uses passlib's pure-python
# pbkdf2_sha256 scheme and does not need the [bcrypt] extra.

python-multipart
